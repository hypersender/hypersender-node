# SmsSendMessageRequest

## Example Usage

```typescript
import { SmsSendMessageRequest } from "hypersender/models/operations";

let value: SmsSendMessageRequest = {
  to: "+12025550147",
  content: "Acme code: 482913",
  requestId: "order-1042-otp",
  scheduleSendAt: "2026-11-20 09:30:00",
  maxSendAttempts: 3,
  messageExpirationSeconds: 600,
};
```
